EJEMPLO FUNCIONAL BÁSICO
================================================================================

```javascript
import { useState } from "react";
import { cn } from "@/lib/utils";
import { Button } from "@/components/ui/button";
import { Card, CardContent } from "@/components/ui/card";

interface MapBlockProps {
  id: string;
  name: string;
  emoji: string;
  onClick?: (id: string) => void;
}

export function MapBlock({ id, name, emoji, onClick }: MapBlockProps) {
  const [selected, setSelected] = useState(false);

  return (
    <Card
      className={cn(
        "cursor-pointer transition-all animate-fade-in",
        selected && "border-primary bg-primary-soft"
      )}
      onClick={() => {
        setSelected(!selected);
        onClick?.(id);
      }}
    >
      <CardContent className="p-4">
        <div className="flex items-center gap-3">
          <span className="text-2xl">{emoji}</span>
          <div className="min-w-0 flex-1">
            <p className="truncate font-semibold">{name}</p>
            <p className="text-xs text-muted-foreground">{id}</p>
          </div>
        </div>
        <Button className="w-full mt-3 rounded-xl">
          Seleccionar
        </Button>
      </CardContent>
    </Card>
  );
}
```

Este ejemplo demuestra:
- useState para estado local (selected)
- cn() para clases condicionales
- Colores del sistema (primary, primary-soft)
- Animación fade-in
- Responsive design con truncate
- Componentes shadcn/ui (Card, CardContent, Button)
